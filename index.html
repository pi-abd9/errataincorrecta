let globalSharedHat = []; // Holds the single master melody sequence function createVoiceLine(speedMs, startingOffset) { const osc = audioCtx.createOscillator(); osc.type = 'sine'; const voiceGain = audioCtx.createGain(); voiceGain.gain.setValueAtTime(0.03, audioCtx.currentTime); let cardIndex = startingOffset; // Stagger where each voice starts in the sequence osc.frequency.setValueAtTime(globalSharedHat[cardIndex % 25], audioCtx.currentTime); osc.connect(voiceGain); voiceGain.connect(masterGain); osc.start(); activeVoices.push(osc); const interval = setInterval(() => { cardIndex++; const nextFreq = globalSharedHat[cardIndex % 25]; osc.frequency.exponentialRampToValueAtTime(nextFreq, audioCtx.currentTime + 0.8); }, speedMs); sequenceIntervals.push(interval); } function startDuchampChoir() { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); masterGain = audioCtx.createGain(); masterGain.gain.setValueAtTime(0.12, audioCtx.currentTime); masterGain.connect(audioCtx.destination); globalSharedHat = generateDuchampHat(); // Generate one single sequence for the entire choir // Microtonal speed differences cause a physical phasing warp createVoiceLine(2000, 0); // Master voice starting at index 0 createVoiceLine(2040, 5); // Slightly slower, starting 5 steps ahead createVoiceLine(2080, 10); // Even slower, starting 10 steps ahead } }
